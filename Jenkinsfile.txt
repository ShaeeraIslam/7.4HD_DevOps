pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                bat 'pip install -r requirements.txt'
            }
        }

        stage('Test') {
            steps {
                bat 'pytest --maxfail=1 --disable-warnings -q'
            }
        }

        stage('Code Quality') {
            steps {
                bat 'flake8 app.py'
            }
        }

        stage('Security') {
            steps {
                bat 'bandit -r app.py'
            }
        }

        stage('Deploy') {
            steps {
                bat 'start /B python app.py'
            }
        }

        stage('Release') {
            steps {
                echo 'Simulating release to production environment...'
            }
        }

        stage('Monitoring') {
            steps {
                bat 'curl http://localhost:5000/health'
            }
        }
    }
}
